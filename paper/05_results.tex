\subsection*{Architectures \& Positional Embeddings}
\begin{table}[h!]
    \centering
    \begin{tabular}{lcccccccc}
        \hline
        Model                                        & Val Loss & Perplexity & Distinct-1 & Distinct-2 &   BERT & ROUGE-1 & ROUGE-2\\
        \hline
        Bigram 0.5M                                  &    6.125 &   1869.72  &     0.6549 &     0.9900 & 0.8153 &  0.1251 & 0.0141 \\
        GRU 1.9M                                     &    4.079 &            &            &            &        &         &  \\
        \makecell[l]{Transformer\\sinusoidalPE 0.7M} &    4.170 &      65.43 &     0.5097 &     0.9155 & 0.8264 &  0.1346 & 0.0165 \\
        \hline
    \end{tabular}
    \caption{Evaluation metrics for different model architectures (all L2).}
\end{table}
\begin{itemize}
    \item compare:
    \begin{itemize}
        \item bigram with sinusoidal positional embedding
        \item bigram with RoPE
        \item GRU with sinusoidal positional embedding
        \item GRU with RoPE
        \item transformer with sinusoidal positional embedding
        \item transformer with RoPE (Rotary Positional Embedding)
    \end{itemize}
    \item results: transformer with RoPE yields the best results
    \item therefore we will use transformer with RoPE in the further experiments
\end{itemize}

\begin{itemize}
    \item Compare:
    \begin{itemize}
        \item with weight initialization
        \item without weight initialization
    \end{itemize}
    \item for transformer with sinusoidal positional embedding \& with RoPE
    \item weight initialization improves results
    \item therefore we will weight initialization from now on
\end{itemize}
